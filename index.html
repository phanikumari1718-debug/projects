/* Application Layer: Finite State Machine Dispatcher */
while (1) {
    keypad_scan();
    scheduler_tick();

    switch (current_state) {
        case STATE_IDLE:
            lcd_print("READY -- PRESS START");
            if (key_pressed(KEY_START)) {
                timer_load(480); /* 8 min wash */
                pwm_set_duty(40); /* 40% PWM duty */
                current_state = STATE_WASH;
            }
            break;

        case STATE_WASH:
            lcd_print_format("WASH %02d:%02d", rem_m, rem_s);
            if (key_pressed(KEY_STOP)) { enter_paused(); break; }
            if (timer_expired())      { transition_to(STATE_RINSE); }
            break;
    }
}

/* Hardware Timer Interrupt Handler (10ms Periodic Tick) */
volatile uint32_t remaining_s = 0;
volatile uint8_t  tick_10ms   = 0;

void TIM2_IRQHandler(void) {
    if (TIM2->SR & TIM_SR_UIF) {
        TIM2->SR &= ~TIM_SR_UIF; /* Clear ISR flag */
        tick_10ms = 1;
        static uint8_t sub_counter = 0;
        if (++sub_counter >= 100) { /* 100 x 10ms = 1 sec */
            sub_counter = 0;
            if (remaining_s > 0) remaining_s--;
        }
    }
}

% Extended Kalman Filter State Estimation Matrix
x_hat = A * x_prev + B * I_pack; % Predict state vector [SOC, V_rc]
P = A * P_prev * A' + Q;          % Covariance update

% Compute Kalman Gain
K = P * H' * inv(H * P * H' + R);
x_hat = x_hat + K * (V_measured - V_ocv(SOC)); % Correct state estimate

/* 12-Bit ADC Sampling with Moving Average Filter */
uint16_t read_engine_temp_adc(void) {
    ADC1->CR2 |= ADC_CR2_SWSTART; /* Trigger ADC conversion */
    while (!(ADC1->SR & ADC_SR_EOC)); /* Wait conversion */
    uint16_t raw_adc = ADC1->DR;
    
    /* 8-sample Exponential Moving Average */
    static uint32_t filtered = 0;
    filtered = (filtered * 7 + raw_adc) / 8;
    return (uint16_t)filtered;
}

// 8-Tap Direct Form FIR Low-Pass Filter Module
module fir_filter_8tap (
    input  wire        clk,
    input  wire        rst_n,
    input  wire signed [15:0] sample_in,
    output reg  signed [31:0] filter_out
);
    // Shift register pipeline for 8 taps
    reg signed [15:0] shift_reg [0:7];
    integer i;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            for (i=0; i<8; i=i+1) shift_reg[i] <= 0;
            filter_out <= 0;
        end else begin
            shift_reg[0] <= sample_in;
            for (i=1; i<8; i=i+1) shift_reg[i] <= shift_reg[i-1];
        end
    end
endmodule
