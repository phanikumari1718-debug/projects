wash_cycle_fsm.c — State Machine Dispatcher C99 Bare-Metal 
while (1) {
    keypad_scan();
    scheduler_tick();

    switch (current_state) {
        case STATE_IDLE:
            lcd_print("READY -- PRESS START");
            if (key_pressed(KEY_START)) {
                timer_load(480); /* 8 min wash */
                pwm_set_duty(40);  /* 40% PWM duty */
                current_state = STATE_WASH;
            }
            break;

        case STATE_WASH:
            lcd_print_format("WASH %02d:%02d", rem_m, rem_s);
            if (key_pressed(KEY_STOP)) { enter_paused(); break; }
            if (timer_expired())      { transition_to(STATE_RINSE); }
            break;
    }
}

adc_temp_sampler.c — Filtered ADC Acquisition ARM Cortex-M 
uint16_t read_engine_temp_adc(void) {
    ADC1->CR2 |= ADC_CR2_SWSTART;
    while (!(ADC1->SR & ADC_SR_EOC));
    uint16_t raw_adc = ADC1->DR;
    
    /* 8-sample exponential moving average */
    static uint32_t filtered = 0;
    filtered = (filtered * 7 + raw_adc) / 8;
    return (uint16_t)filtered;
}
